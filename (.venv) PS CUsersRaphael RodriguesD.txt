class GraficoObras(QWidget):

    def __init__(self, parent=None):

        super().__init__(parent)

        self.dados = []
        self.tema_escuro = False

        self.setMinimumHeight(370)

        self.setSizePolicy(
            QSizePolicy.Policy.Expanding,
            QSizePolicy.Policy.Fixed
        )

    def definir_dados(self, dados):

        self.dados = dados
        self.update()

    def atualizar_tema(self, escuro):

        self.tema_escuro = bool(escuro)
        self.update()

    def paintEvent(self, evento):

        painter = QPainter(self)

        painter.setRenderHint(
            QPainter.RenderHint.Antialiasing
        )

        if self.tema_escuro:

            cor_texto = QColor("#E2E8F0")
            cor_secundaria = QColor("#A8B6C8")
            cor_linha = QColor("#354254")

            cor_frente = QColor("#438CCB")
            cor_lateral = QColor("#28669D")
            cor_topo = QColor("#82B9E8")

        else:

            cor_texto = QColor("#263445")
            cor_secundaria = QColor("#64748B")
            cor_linha = QColor("#E1E8F0")

            cor_frente = QColor("#347FC0")
            cor_lateral = QColor("#205B91")
            cor_topo = QColor("#78B5E5")

        largura = self.width()
        altura = self.height()

        if not self.dados:

            painter.setPen(cor_secundaria)

            painter.drawText(
                self.rect(),
                Qt.AlignmentFlag.AlignCenter,
                "Nenhum dado disponÃ­vel"
            )

            painter.end()
            return

        quantidade_colunas = len(self.dados)

        maior_valor = max(
            valor for _, valor in self.dados
        )

        if maior_valor <= 0:
            painter.end()
            return

        # Ãrea do grÃ¡fico

        margem_esquerda = 35
        margem_direita = 30

        area_largura = (
            largura - margem_esquerda - margem_direita
        )

        topo_grafico = 55
        base_grafico = altura - 65

        altura_disponivel = base_grafico - topo_grafico

        espacamento = (
            area_largura / quantidade_colunas
        )

        largura_coluna = min(
            58,
            espacamento * 0.48
        )

        profundidade = 12

        # Linha de base

        painter.setPen(cor_linha)

        painter.drawLine(
            margem_esquerda,
            base_grafico,
            largura - margem_direita,
            base_grafico
        )

        painter.setFont(
            QFont("Segoe UI", 8)
        )

        for indice, (obra, quantidade) in enumerate(self.dados):

            centro_x = (
                margem_esquerda
                + espacamento * indice
                + espacamento / 2
            )

            x = centro_x - largura_coluna / 2

            altura_coluna = (
                quantidade / maior_valor
            ) * altura_disponivel * 0.82

            y = base_grafico - altura_coluna

            # Sombra da coluna

            painter.setPen(Qt.PenStyle.NoPen)

            painter.setBrush(
                QColor(0, 0, 0, 25)
            )

            painter.drawRoundedRect(
                int(x + 5),
                int(y + 8),
                int(largura_coluna),
                int(altura_coluna),
                5,
                5
            )

            # Frente da coluna

            painter.setBrush(cor_frente)

            painter.drawRoundedRect(
                int(x),
                int(y),
                int(largura_coluna),
                int(altura_coluna),
                4,
                4
            )

            # Lateral 3D

            lateral = QPolygon([
                QPoint(
                    int(x + largura_coluna),
                    int(y)
                ),
                QPoint(
                    int(x + largura_coluna + profundidade),
                    int(y - profundidade)
                ),
                QPoint(
                    int(x + largura_coluna + profundidade),
                    int(base_grafico - profundidade)
                ),
                QPoint(
                    int(x + largura_coluna),
                    int(base_grafico)
                )
            ])

            painter.setBrush(cor_lateral)
            painter.drawPolygon(lateral)

            # Topo 3D

            topo = QPolygon([
                QPoint(int(x), int(y)),
                QPoint(
                    int(x + profundidade),
                    int(y - profundidade)
                ),
                QPoint(
                    int(x + largura_coluna + profundidade),
                    int(y - profundidade)
                ),
                QPoint(
                    int(x + largura_coluna),
                    int(y)
                )
            ])

            painter.setBrush(cor_topo)
            painter.drawPolygon(topo)

            # Quantidade acima da coluna

            painter.setPen(cor_texto)

            painter.setFont(
                QFont("Segoe UI", 9, QFont.Weight.Bold)
            )

            painter.drawText(
                int(x - 12),
                int(y - profundidade - 12),
                int(largura_coluna + profundidade + 24),
                18,
                Qt.AlignmentFlag.AlignCenter,
                str(quantidade)
            )

            # Nome da obra abaixo da coluna

            nome = str(obra).strip()

            if len(nome) > 15:
                nome = nome[:13] + "..."

            painter.setPen(cor_secundaria)

            painter.setFont(
                QFont("Segoe UI", 8)
            )

            painter.drawText(
                int(centro_x - espacamento / 2),
                base_grafico + 18,
                int(espacamento),
                35,
                Qt.AlignmentFlag.AlignHCenter
                | Qt.AlignmentFlag.AlignTop,
                nome
            )

        painter.end()

class TrabalhadorAnalise(QThread):

    progresso = Signal(str, int, int, int, int)
    concluido = Signal(object)
    erro = Signal(str)

    def __init__(self, caminhos):
        super().__init__()
        self.caminhos = list(caminhos)

    def executar_progresso(
        self,
        nome_arquivo,
        indice_arquivo,
        total_arquivos,
        pagina_atual,
        total_paginas
    ):
        self.progresso.emit(
            nome_arquivo,
            indice_arquivo,
            total_arquivos,
            pagina_atual,
            total_paginas
        )

    def run(self):
        try:
            resultado = analisar_varios_pdfs(
                self.caminhos,
                progresso_callback=self.executar_progresso
            )

            self.concluido.emit(resultado)

        except Exception as erro:
            self.erro.emit(str(erro))

    # --------------------------------------------------
    # ANIMACAO DE TRANSPARENCIA INDIVIDUAL
    # --------------------------------------------------

    def limpar_animacoes(self):

        if self.grupo_animacao is not None:
            self.grupo_animacao.stop()
            self.grupo_animacao = None

        for widget, efeito in self.itens_animacao:

            widget.setGraphicsEffect(None)

        self.itens_animacao.clear()

    def iniciar_transicao_resultados(self):

        self.limpar_animacoes()

        self.area_rolagem.verticalScrollBar().setValue(0)

        self.definir_opcoes_pos_analise(True)

        layout = self.area_rolagem.widget().layout()

        if layout:
            layout.activate()

        self.area_rolagem.viewport().update()

        # A animacao comeca antes do proximo desenho da tela.
        self.executar_animacao_transparencia()

    def executar_animacao_transparencia(self):

        elementos = [
            self.titulo_resumo,

            self.card_total,
            self.card_identificadas,
            self.card_revisar,
            self.card_pendentes,

            self.cartao_resumo_obras,
            self.cartao_filtro,

            self.titulo_tabela,
            self.tabela,

            self.botao_marcar,
            self.botao_desmarcar,
            self.botao_gerar
        ]

        self.grupo_animacao = QSequentialAnimationGroup(self)

        for widget in elementos:

            if not widget.isVisible():
                continue

            # Cada elemento recebe seu proprio efeito
            # de opacidade, sem cobertura preta ou solida.
            efeito = QGraphicsOpacityEffect(widget)

            efeito.setOpacity(0.0)

            widget.setGraphicsEffect(efeito)

            self.itens_animacao.append(
                (widget, efeito)
            )

            animacao = QPropertyAnimation(
                efeito,
                b"opacity",
                self.grupo_animacao
            )

            animacao.setDuration(320)

            animacao.setStartValue(0.0)
            animacao.setEndValue(1.0)

            animacao.setEasingCurve(
                QEasingCurve.Type.OutCubic
            )

            self.grupo_animacao.addAnimation(animacao)

        self.grupo_animacao.finished.connect(
            self.finalizar_animacao_transparencia
        )

        self.grupo_animacao.start()

    def finalizar_animacao_transparencia(self):

        for widget, efeito in self.itens_animacao:

            widget.setGraphicsEffect(None)

        self.itens_animacao.clear()

        self.grupo_animacao = None

    def cancelar_animacao_transparencia(self):

        self.limpar_animacoes()

    # --------------------------------------------------
    # IDENTIFICACAO DAS OBRAS
    # --------------------------------------------------

    def obter_obras_associacao(self, associacao):

        obras = set()

        for veiculo in associacao.get("veiculos", []):

            obra = str(veiculo.get("obra", "")).strip()
            codigo = str(veiculo.get("codigo_obra", "")).strip()

            if codigo or obra:

                identificacao = (
                    f"{codigo or 'SEM CODIGO'} - "
                    f"{obra or 'OBRA NAO IDENTIFICADA'}"
                )

                obras.add(identificacao)

        if not obras:
            obras.add("SEM OBRA IDENTIFICADA")

        return obras

    # --------------------------------------------------
    # ATUALIZAR CARDS
    # --------------------------------------------------

    def atualizar_cards(self):

        identificadas = sum(
            1 for item in self.associacoes
            if item["status"] == "identificado"
        )

        revisar = sum(
            1 for item in self.associacoes
            if item["status"] == "revisar"
        )

        pendentes = sum(
            1 for item in self.associacoes
            if item["status"] == "pendente"
        )

        self.card_total.valor.setText(str(len(self.associacoes)))
        self.card_identificadas.valor.setText(str(identificadas))
        self.card_revisar.valor.setText(str(revisar))
        self.card_pendentes.valor.setText(str(pendentes))

    # --------------------------------------------------
    # RESUMO DE FATURAS POR OBRA
    # --------------------------------------------------

    def atualizar_resumo_obras(self):

        contagem = Counter()

        for associacao in self.associacoes:

            for obra in self.obter_obras_associacao(associacao):
                contagem[obra] += 1

        if not contagem:
            self.resumo_obras.setText(
                "Nenhuma obra identificada."
            )
            return

        linhas = []

        for obra, quantidade in sorted(contagem.items()):
            linhas.append(f"- {obra}: {quantidade} fatura(s)")

        linhas.append("")
        linhas.append(f"TOTAL: {len(self.associacoes)} fatura(s)")

        self.resumo_obras.setText("\n".join(linhas))

    # --------------------------------------------------
    # PREENCHER FILTRO
    # --------------------------------------------------

    def preencher_filtro_obras(self):

        self.lista_obras.blockSignals(True)
        self.lista_obras.clear()

        contagem = Counter()

        for associacao in self.associacoes:

            for obra in self.obter_obras_associacao(associacao):
                contagem[obra] += 1

        for obra, quantidade in sorted(contagem.items()):

            item = QListWidgetItem(
                f"{obra}  ({quantidade})"
            )

            item.setData(
                Qt.ItemDataRole.UserRole,
                obra
            )

            item.setFlags(
                item.flags() | Qt.ItemFlag.ItemIsUserCheckable
            )

            item.setCheckState(
                Qt.CheckState.Unchecked
            )

    def aplicar_filtros(self):

        dados = self.dados.copy()

        obra = self.filtro_obra.currentText()

        responsavel = self.filtro_responsavel.currentText()

        placa = self.filtro_placa.text().strip().upper()

        placa = placa.replace("-", "").replace(" ", "")

        if obra != "Todas as obras":

            dados = dados[
                dados["OBRA"] == obra
            ]

        if responsavel != "Todos os responsÃ¡veis":

            dados = dados[
                dados["RESPONSAVEL"] == responsavel
            ]

        if placa:

            dados = dados[
                dados["PLACA"].astype(str).str.upper().str.contains(
                    placa,
                    na=False
                )
            ]

        self.atualizar_tabela(dados)

    def atualizar_tabela(self, dados=None):

        if dados is None:
            dados = self.dados

        self.tabela.setRowCount(len(dados))

        for linha, (_, registro) in enumerate(
            dados.iterrows()
        ):

            placa = str(registro["PLACA"])

            if len(placa) == 7:
                placa = placa[:3] + "-" + placa[3:]

            valores = [
                placa,
                registro["ATIVO"],
                registro["OBRA"],
                registro["CODIGO DA OBRA"],
                registro["RESPONSAVEL"]
            ]

            for coluna, valor in enumerate(valores):

                item = QTableWidgetItem(
                    str(valor) if valor is not None else ""
                )

                item.setTextAlignment(
                    Qt.AlignmentFlag.AlignCenter
                )

                self.tabela.setItem(
                    linha,
                    coluna,
                    item
                )
